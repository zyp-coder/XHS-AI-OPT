## 1. content.js·`parseNoteStats(panel)`

文件：`extension/content.js`，**新增**函数，紧挨 `extractProfileFull`（≈ line `3932`）后。

```js
/** 单页笔记的统计：互动三连（点赞 / 收藏 / 评论）；失败返 null + 错误原因（不抛）
 *  优先 aria-label，DOM 兜底：
 *   - aria-label 里以 '点赞' / '赞' / '收藏' / '评论' / '回复' 起头
 *   - 兜底：节点 [+所有 .count / .stat / .num 子节点] 抓 '数字' 部分
 *  返回 { likes:number, collects:number, comments:number, raw } —— 字段全填不到返 null */
function parseNoteStats(panel) {
  if (!panel) return null;
  const out = { likes: 0, collects: 0, comments: 0, raw: '' };
  const txt = (el) => (el ? String(el.textContent || el.getAttribute('aria-label') || '').replace(/\s+/g, ' ').trim() : '');
  const num = (s) => {
    if (s == null) return 0;
    const str = String(s).replace(/[^\d.万]/g, '');
    const m = str.match(/(\d+(?:\.\d+)?)(万)?/);
    if (!m) return 0;
    let n = parseFloat(m[1]);
    if (m[2] === '万') n *= 10000;
    return Math.round(n);
  };
  const elFromAria = (key) => panel.querySelector(
    '[aria-label*="' + key + '"], [aria-label*="' + key.toUpperCase() + '"]'
  );
  /* --- ① 优先 aria-label --- */
  let cnt = 0;
  for (const k of ['点赞', '赞']) {
    const el = elFromAria(k);
    const v = num(txt(el));
    if (v > 0) { out.likes = v; cnt++; break; }
  }
  for (const k of ['收藏']) {
    const el = elFromAria(k);
    const v = num(txt(el));
    if (v > 0) { out.collects = v; cnt++; break; }
  }
  for (const k of ['评论', '回复']) {
    const el = elFromAria(k);
    const v = num(txt(el));
    if (v > 0) { out.comments = v; cnt++; break; }
  }
  /* --- ② 兜底：拿 .count/.stat 列表，按出现的顺序配 likes/collects/comments --- */
  if (cnt < 3) {
    const cells = Array.from(panel.querySelectorAll('[class*="count"], [class*="stat"], [class*="num"]'))
      .map(el => num(txt(el))).filter(n => n > 0);
    if (cells.length >= 1 && !out.likes) out.likes = cells[0];
    if (cells.length >= 2 && !out.collects) out.collects = cells[1];
    if (cells.length >= 3 && !out.comments) out.comments = cells[2];
  }
  /* --- ③ 兜不住：基本没抓到，返 null + 错误 --- */
  if (!out.likes && !out.collects && !out.comments) {
    return { error: '没抓到任何数字（DOM 没出 .count，或者三连 aria-label 全空）' };
  }
  out.raw = JSON.stringify({ likes: out.likes, collects: out.collects, comments: out.comments });
  return out;
}
```

## 2. background.js·stats 路由

文件：`extension/background.js`，**新增** message handler：

```js
// 把 content.js 抓到的 stats 写进 chrome.storage.local['note_stats']
async function statsWrite(payload, sender) {
  if (!payload || !payload.id || !payload.stats) return { ok: false };
  const key = 'note_stats';
  try {
    const d = await chrome.storage.local.get(key);
    const m = d[key] || {};
    m[payload.id] = {
      likes: Number(payload.stats.likes || 0),
      collects: Number(payload.stats.collects || 0),
      comments: Number(payload.stats.comments || 0),
      raw: String(payload.stats.raw || ''),
      fetchedAt: Date.now(),
    };
    await chrome.storage.local.set({ [key]: m });
    return { ok: true };
  } catch (e) { return { ok: false, error: String(e.message || e) }; }
}
function pushStats(url, stats) {
  if (!url || !stats) return;
  const m = String(url).match(/\/explore\/([^?#]+)/);
  if (!m) return;
  chrome.runtime.sendMessage({ action: 'writeNoteStats', id: m[1], stats }).catch(() => {});
}
chrome.runtime.onMessage.addListener((req, sender, sendResponse) => {
  if (req && req.action === 'writeNoteStats') return statsWrite(req, sender).then(r => sendResponse(r));
});
```

加进 `manifest.json`（**不改！** —— message 协议不需要 manifest key；这条就用现有的 `externally_connectable` / 默认就能用）：

- content.js / popup.js 用 `chrome.runtime.sendMessage({ action: 'writeNoteStats', id, stats })` 即可。

## 3. popup.js·`npStatsWrite` 三档配对

文件：`extension/popup.js`，**新增**一节：

```js
const NP_STATS_KEY = 'note_stats';
async function npStatsReadAll() {
  try { const d = await chrome.storage.local.get(NP_STATS_KEY); return d[NP_STATS_KEY] || {}; }
  catch (_) { return {}; }
}
async function npStatsRead(noteId) {
  const all = await npStatsReadAll();
  return all[noteId] || null;
}

/** npStatsWriteByTitle(list, noteStatsAll)
 *  - list = npNotesAll()
 *  - noteStatsAll = npStatsReadAll() 出来的 { [noteId]: stats }
 *  对每条档案条目，按「小红书那一条」URL 中的 explore id 反查 stats
 *  三档模糊匹配（按 title 匹配 explore id）：
 *    ① 完全相等
 *    ② 标题去尾 3 字（运营上常见去「（合集）」、「的呀」一类装饰）
 *    ③ 标题包含
 *  返回 { hit, miss }，hint 给 UI 看「抓到 N · 配上 M」 */
async function npStatsWriteByTitle(list, noteStatsAll) {
  const hit = [];
  const miss = [];
  // 反查表：known titles → exploreId；用同 stats 全表的 id 反查
  // 这里走 exploreId → 标题 的索引，先把 noteStatsAll key 视作 exploreId
  const idList = Object.keys(noteStatsAll);
  const statsById = noteStatsAll;
  // 遍历 np_notes，找到该条 exploreId
  for (const n of list) {
    if (!n || n.status !== '已发') continue;                   // 只对已发档案
    const expId = await npExploreIdByUrl(n.url);
    if (!expId) { miss.push({ id: n.id, reason: '档案没记 url' }); continue; }
    const s = statsById[expId];
    if (!s) { miss.push({ id: n.id, reason: '还没抓主页', expId }); continue; }
    // 落 stats 进档案条
    n.stats = s;
    n.statsAt = s.fetchedAt;
    hit.push(n.id);
  }
  return { hit, miss };
}

/** 给某篇档案找回 explore id —— 优先读 `n.url`，没有就走 npNotesSync 时的产物。 */
async function npExploreIdByUrl(u) {
  if (!u) return '';
  const m = String(u).match(/\/explore\/([^?#]+)/);
  return m ? m[1] : '';
}
```

### 3.1 回写档案

```js
async function npStatsAdopt() {
  // 每次启动跑：把 note_stats 配回档案
  const list = await npNotesAll();
  const stats = await npStatsReadAll();
  const { hit, miss } = await npStatsWriteByTitle(list, stats);
  if (hit.length) {
    await npNotesSync(async () => {
      await chrome.storage.local.set({ [NP_NOTES_KEY]: list });
    });
  }
  return { hit: hit.length, miss: miss.length, sample: miss.slice(0, 5) };
}
```

```js
document.getElementById('btnStatsRefetch')?.addEventListener('click', npStatsRefetchAll);
document.getElementById('btnStatsRefetchOne')?.addEventListener('click', npStatsRefetchOne);

async function npStatsRefetchAll() {
  const list = await npNotesAll();
  const exIds = [];
  for (const n of list) {
    if (n.status !== '已发') continue;
    const e = await npExploreIdByUrl(n.url);
    if (e) exIds.push({ id: e, title: n.title });
  }
  if (!exIds.length) return toast('⚠️ 没找到一篇已发带 url 的档案');
  // 用 chromnium debugger 或 open note —— 走 opener：
  // 走 popup 里 _statsOpenerTabs：依次打开每个 exploreId，每开一个新 tab 等它 'complete' 再抓 stats 一次
  let ok = 0, bad = 0;
  for (const e of exIds) {
    const u = 'https://www.xiaohongshu.com/explore/' + e.id;
    const tab = await chrome.tabs.create({ url: u, active: false });
    await new Promise((r, j) => {
      const t = setTimeout(() => j(new Error('time out ' + e.id)), 8000);
      const fn = (tabId, ci) => {
        if (tabId !== tab.id || ci !== 'complete') return;
        clearTimeout(t); chrome.tabs.onUpdated.removeListener(fn);
        setTimeout(async () => {
          const [, r] = await chrome.scripting.executeScript({
            target: { tabId: tab.id },
            files: ['content.js'],
          });
          const stats = (r && r[0] && r[0].result) || null;
          if (stats && !stats.error) {
            await npStatsRead(); // background auto persist via writeNoteStats
            ok++;
          } else bad++;
          chrome.tabs.remove(tab.id).catch(() => {});
          r();
        }, 1500);
      };
      chrome.tabs.onUpdated.addListener(fn);
    });
  }
  await npStatsAdopt();
  toast(ok ? `✅ 抓主页完成 ${ok}/${ok + bad}` : `❌ 一次都没抓到（可能没装 content script，或者小红书改版了）`);
}

async function npStatsRefetchOne() {
  const cur = window._npCurReviewNote;       // 复盘分析页当前查看的篇
  if (!cur) return toast('⚠️ 先在已发复盘里点一篇');
  const e = await npExploreIdByUrl(cur.url);
  if (!e) return toast('⚠️ 这篇没记 url，先回档案补');
  // 走 npStatsRefetchAll 单条版
  const tab = await chrome.tabs.create({ url: 'https://www.xiaohongshu.com/explore/' + e, active: false });
  await new Promise((r) => {
    const fn = (tabId, ci) => {
      if (tabId !== tab.id || ci !== 'complete') return;
      chrome.tabs.onUpdated.removeListener(fn);
      setTimeout(async () => {
        const [, rr] = await chrome.scripting.executeScript({
          target: { tabId: tab.id }, files: ['content.js'],
        });
        const stats = (rr && rr[0] && rr[0].result) || null;
        if (stats && !stats.error) await npStatsAdopt();
        chrome.tabs.remove(tab.id).catch(() => {});
        toast(stats ? '✅ 已抓' : (stats && stats.error) || '❌ 没抓到');
        r();
      }, 1500);
    };
    chrome.tabs.onUpdated.addListener(fn);
  });
}
```

### 5.1 `_check_stats.py`

文件：`scripts/dev/_check_stats.py`，新建：

```python
#!/usr/bin/env python3
"""A5 stats 自检：note_stats.json + np_notes.json 配对率"""
import json, pathlib

def main():
    base = pathlib.Path(sys.argv[1] if len(sys.argv) > 1 else '.')
    stats_raw = json.loads((base / 'note_stats.json').read_text(encoding='utf-8'))
    notes = json.loads((base / 'np_notes.json').read_text(encoding='utf-8'))
    if isinstance(stats_raw, list):
        stats = stats_raw
    else:
        stats = [{'id': k, **v} for k, v in (stats_raw if isinstance(stats_raw, dict) else {}).items()]
    published = [n for n in notes if isinstance(n, dict) and n.get('status') == '已发']
    print(f"📦 note_stats 条数: {len(stats)}")
    print(f"📚 np_notes 已发条数: {len(published)}")
    hit = 0
    for n in published:
        url = n.get('url', '')
        m = __import__('re').search(r'/explore/([^?#]+)', url)
        if not m: continue
        eid = m[1]
        for s in stats:
            if s.get('id') == eid:
                n.setdefault('_stats', s); hit += 1; break
    print(f"✅ 配上 stats: {hit} / {len(published)}")

if __name__ == '__main__':
    main()
```

## 7. 注意事项·别踩坑

1. **不抓别人的号**：`statsWrite` 只入档案条自己有的笔记——`extractGuestProfile` 那侧**不写**，只 profile（自己）会走这条。
2. **不开 `chrome.debugger`**：走 `tabs.create` + `scripting.executeScript` 即可；CDP 协议被风控点名，合规边界不能跨。
3. **抓主页不走 manifest 加 permission**：用现有的 `chrome.tabs` / `chrome.scripting`，不加 manifest key。
4. **「🔄 重抓」开 tab 的 active = false**：不让用户在窗口里跳；开完即关。
5. **每条 exploreUrl 间隔 1.5 秒**：`chrome.tabs.create` 之后 `setTimeout(fn, 1500)`，让小红书有时间把 stats DOM 渲出来——少开 zhenlvqi 的反爬点名。
6. **stats 7 天就过期**：UI 标「🕗 7 天前的数据」，自动重新抓；不主动提示用户。
7. **profile 页 `extractProfileFull` 失败 `notReady` 状态保留**：跟 `notProfile / notReady` 三态走，stats 抓不到不算错。
