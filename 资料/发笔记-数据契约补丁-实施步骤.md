### 1.2 队列独立 key：`note_queue`

```js
const NP_NOTE_QUEUE_KEY = 'note_queue';
```

```js
/** 老的「待备」算法沿用：A4 新加的这套是「把素材重组成可重看的集合」 */
async function npAsGapsForCollections() {
  const collections = await npCollectionsRead();
  const needs = await npAsNeedsRead();
  const out = [];
  for (const c of collections) {
    if (c.source) {
      const m = c.source.match(/^(sell|kb)\|([^|]+)$/);
      if (!m) { out.push({ id: c.id, missing: true }); continue; }
      const need = needs.find(n => n.source === c.source);
      if (!need) { out.push({ id: c.id, missing: true }); continue; }
      out.push({ id: c.id, missing: false, needId: need.id });
    }
  }
  return out;
}
```

### 3.3 连播状态 / 单段播放

视频段之间 `▶` button 用 `chrome.tabs.create` 走「文件协议」/「data URL」，**不另存**：

```js
async function npClipPlay(clipId) {
  const clips = await npClipsRead();
  const c = clips.find(x => x.id === clipId);
  if (!c) return;
  if (c.kind === 'image') return window.open(c.ref, '_blank');
  // 视频段：用 video tag + URL.createObjectURL(blob)
```

### 5.1 `note_minds`

文件：`extension/popup.js`：

```js
const NP_MINDS_KEY = 'note_minds';
async function npMindsRead() {
  try { const d = await chrome.storage.local.get(NP_MINDS_KEY); return d[NP_MINDS_KEY] || {}; }
  catch (_) { return {}; }
}
async function npMindsWrite(map) {
  try { await chrome.storage.local.set({ [NP_MINDS_KEY]: map }); } catch (_) {}
}
async function npMindsUpdate(noteId, mind) {
  const map = await npMindsRead();
  const flat = npMindFlat(mind);                                       // 把 nodes 拍平成 'a|b|c' 的纯文本
  if (flat) map[noteId] = { flat, at: Date.now() };
  else delete map[noteId];
  await npMindsWrite(map);
}
function npMindFlat(m) {
  if (!m || !Array.isArray(m.nodes)) return '';
  const out = [];
  function walk(nodes, prefix) {
    for (const n of nodes) {
      const text = String(n.text || '').trim();
      if (text) out.push(prefix + text);
      if (Array.isArray(n.sub)) walk(n.sub, prefix + '- ');
    }
  }
  walk(m.nodes, '');
  return out.join('\n');
}
```

## 6. 一次走通的 _sv 迁移

A1 是 `_sv:1`；A2~A6 完成之后**整个 schema 升到 `_sv:2`**。A1 写过的 `npNotesMigrate` 改成：

```js
const NP_SCHEMA_VER = 2;
async function npNotesMigrate() {
  await npNotesSync(async () => {
    const list = await npNotesAll();
    let dirty = false;
    for (const n of list) {
      if (!n) continue;
      if (n._sv && n._sv >= NP_SCHEMA_VER) continue;
      // A1 字段
      Object.assign(n, npArchiveCoerce(n));
      // A2 字段
      n.queueNotes = Array.isArray(n.queueNotes) ? n.queueNotes : [];   // 该篇在队列里的快照
      // A3 字段
      n.imgMotifs = Array.isArray(n.imgMotifs) ? n.imgMotifs : [];
      n.stagePlan = n.stagePlan || null;
      // A4 字段
      n.collectionIds = Array.isArray(n.collectionIds) ? n.collectionIds : [];
      // A5 字段
      n.stats = n.stats || null;
      n._sv = NP_SCHEMA_VER;
      dirty = true;
    }
    if (dirty) await chrome.storage.local.set({ [NP_NOTES_KEY]: list });
  });
}
```

```python
#!/usr/bin/env python3
"""A2~A6 schema 自检 —— 把 np_notes.json / note_queue.json / np_collections.json 等读完逐项校验"""
import json, sys, pathlib

KEYS = ['np_notes', 'note_queue', 'np_collections', 'np_clips', 'note_stats',
        'note_minds', 'note_dgens', 'note_asms']

def ok(name, cond, detail=''):
    print(('✅ ' if cond else '❌ ') + name + ((': ' + detail) if detail else ''))
    return cond

def main():
    base = pathlib.Path(sys.argv[1] if len(sys.argv) > 1 else '.')
    for k in KEYS:
        p = base / (k + '.json')
        if not p.exists():
            print(f'⚠️  {k}.json 不存在 —— 这一节还没动或者还没数据')
            continue
        raw = json.loads(p.read_text(encoding='utf-8'))
        if k == 'np_notes':
            ok(f'{k} 条数', isinstance(raw, list) and len(raw) > 0, str(len(raw)))
            for n in raw:
                for f in ['dir','kOut','dOut','mind','voiceUrl','dgenDocs','asm']:
                    if f not in n:
                        ok(f'[{n.get("id","?")[:6]}] {f}', False, '缺字段')
                if n.get('_sv', 0) < 2:
                    print(f"⚠️  [{n.get('id','?')[:6]}] _sv={n.get('_sv')!r} < 2，下次写自动迁")
        else:
            ok(f'{k} 是对象或数组', isinstance(raw, (list, dict)))

if __name__ == '__main__':
    main()
```

## 9. 注意事项·别踩坑

1. **A4 双表并存千万别合并**：新 `np_collections / np_clips` 是**展示模型**，旧 `np_needs / np_assets` 是**匹配模型**——双方一起留半年，确认无回归再考虑合并。**目前不合并**。
2. **A5 不动 A1 的 `voiceUrl`**：俩口子别串到一起去，stats 是「抓主页读到的点赞数」，voiceUrl 是「行里听到的配音」，两回事。
3. **A6 三个 key 别写脏数据**：写之前 `npMindFlat(mind)` 的产物是 `'\n'` 连接的多行；空 flat 就 `delete`，别让空字符串进索引。
4. **周排迁移一次性**：`npQueueMigrate` 启动期跑一次（用 `if (cur.length) return 0` 兜底），别加到 `npNotesMigrate` 里——队列是个独立 key。
5. **`_sv` 升 2 的时候别动 A1 那 8 字段默认值**：A1 已迁过的 `_sv:1` 重新 coerce 一遍再 _sv:2 是幂等安全的；不可只升 `_sv` 不补字段。
