### 1.2 `npMindPack` / `npAsmPack` 两个收尾函数

文件：`extension/popup.js`，**放在 `npArchiveData` 上方**（≈ line `7720` 附近）。

```js
/** 思维导图产物收口：删空行 / 截长度，转成档案可存的形状 */
function npMindPack(m) {
  if (!m || typeof m !== 'object') return null;
  const title = String(m.title || '').slice(0, 40);
  const nodes = Array.isArray(m.nodes) ? m.nodes
    .filter(n => n && String(n.text || '').trim())
    .slice(0, 30)
    .map(n => ({
      text: String(n.text || '').slice(0, 80),
      sub: Array.isArray(n.sub)
        ? n.sub.filter(s => s && String(s.text || '').trim()).slice(0, 12).map(s => ({ text: String(s.text || '').slice(0, 80) }))
        : [],
    })) : [];
  if (!nodes.length && !title) return null;
  return { title, nodes };
}

/** 大组装产物收口：bgmId + sections 的双重命名约定 */
function npAsmPack(a) {
  if (!a || typeof a !== 'object') return null;
  const bgmId = String(a.bgmId || '');
  const sections = Array.isArray(a.sections) ? a.sections
    .filter(s => s && String(s.cardKey || '').trim())
    .slice(0, 80)
    .map(s => ({
      cardKey: String(s.cardKey || ''),
      scenes: Array.isArray(s.scenes) ? s.scenes.slice(0, 12).map(sc => ({
        text: String(sc.text || '').slice(0, 200),
        t: Number(sc.t || 0),
      })) : [],
    })) : [];
  if (!sections.length && !bgmId) return null;
  return { bgmId, sections };
}
```

### 1.5 旧档不丢：写入容错

`npArchiveSave` 进函数时加一行护栏：

```js
// A1 老档容错：np_notes 里读到的旧字段（无 dir/kOut/dOut/mind/…）一律当成 null
function npArchiveCoerce(old) {
  if (!old || typeof old !== 'object') return null;
  return Object.assign({}, old, {
    _sv: 1, dir: old.dir || 'old',
    kOut: old.kOut || '', dOut: old.dOut || '',
    mind: old.mind || null, voiceUrl: old.voiceUrl || '',
    voiceError: old.voiceError || null,
    dgenDocs: Array.isArray(old.dgenDocs) ? old.dgenDocs : [],
    asm: old.asm || null,
  });
}
```

`npNotesAll` 调用**就在它下面加**，读 `np_notes` 之后过一遍 coerce 再返回：

```js
async function npNotesAll() {
  try {
    const d = await chrome.storage.local.get(NP_NOTES_KEY);
    const a = d[NP_NOTES_KEY];
    if (!Array.isArray(a)) return [];
    return a.map(npArchiveCoerce).filter(Boolean);
  } catch (_) { return []; }
}
```

> 这样读出来就是**完整的**新档；写到 `chrome.storage.local` 那个动作，得要用户点「💾 保存」或者触发 `npArchiveSave` 才会触发写——读档是只读迁移，老档在第一次写它时**自动就迁移好了**，不会强制回写一遍。

### 4.1 口播 mp4 落 IndexedDB（不走 storage.local）

```js
// 背景页接 manifest mp4（豆包配音 / 即梦数字人 / 本地剪辑）→ 写进 IndexedDB
async function putVoice(key, blob) {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open('xhsVoice', 1);
    req.onsuccess = () => {
      const db = req.result;
      const tx = db.transaction('voice', 'readwrite');
      tx.objectStore('voice').put({ key, blob, at: Date.now() });
      tx.oncomplete = () => resolve(true);
      tx.onerror = () => reject(tx.error);
    };
    req.onerror = () => reject(req.error);
  });
}
async function getVoice(key) {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open('xhsVoice', 1);
    req.onsuccess = () => {
      const db = req.result;
      const tx = db.transaction('voice', 'readonly');
      const g = tx.objectStore('voice').get(key);
      g.onsuccess = () => resolve(g.result ? g.result.blob : null);
      g.onerror = () => reject(g.error);
    };
    req.onerror = () => reject(req.error);
  });
}
```

## 5. 迁移：`npNotesMigrate(runOnce)`

**写一个 idempotent 迁移函数**，应用启动时跑一次：

文件：`extension/popup.js`，**放在 `npNotesAll()` 下面**（≈ line `7710`）。

```js
/** A1 schema 迁移：把 _sv:undefined 的旧档补默认值，已是新档的（_sv>=1）跳过。
 *  幂等 —— 跑 N 次也只跑一次「补字段」，写回时只在字段真缺时才动对象引用。 */
async function npNotesMigrate() {
  let touched = 0;
  await npNotesSync(async () => {
    const list = await npNotesAll();
    let dirty = false;
    const out = list.map(n => {
      if (!n || typeof n !== 'object') return n;
      if (n._sv && n._sv >= 1) return n;        // 已迁过
      dirty = true; touched++;
      return npArchiveCoerce(n);                // 把缺字段补齐
    });
    if (dirty) await chrome.storage.local.set({ [NP_NOTES_KEY]: out });
  });
  return touched;
}
```

调用点（popup 启动入口附近的「一次性」自检里）：

```js
// 启动期一次性迁移
npNotesMigrate().then(n => {
  if (n > 0) console.info('[note-migrate] 补字段 ' + n + ' 条');
});
```

> 这一段**没存进 np_notes 的话不会回写**——上面说「读档 coerce」是给当前会话用，迁移动作只对**还在原位上的老档**做一次真改。
> 不在 np_notes 数组里的（用户从来没存过的档）——本来就不会回到 `np_notes`，不需要迁。

## 6. 验证脚本

### 6.1 `_check_note_archive.py`（自检·直接读 json 备份）

文件：`scripts/dev/_check_note_archive.py`（**新建**，跟现状脚本同路径风格一致）。

> 这份脚本**不进扩展**，是开发机直接读 chrome.storage.local 备份 json（要客户先在扩展 → 设置页导出一份 `np_notes` 备份）。

入口：

```python
#!/usr/bin/env python3
"""A1 schema 自检 —— 把 np_notes.json 读出来逐项校验"""
import json, sys, pathlib

REQ_TOP = {'id','createdAt','updatedAt','status'}
NEW_FIELDS = ['dir','kOut','dOut','mind','voiceUrl','voiceError','dgenDocs','asm']

def ok(name, cond, detail=''):
    print(('✅ ' if cond else '❌ ') + name + ((': ' + detail) if detail else ''))
    return cond

def main():
    p = pathlib.Path(sys.argv[1] if len(sys.argv) > 1 else 'np_notes.json')
    raw = json.loads(p.read_text(encoding='utf-8'))
    arr = raw if isinstance(raw, list) else raw.get('np_notes') or []
    ok('总条数', len(arr) > 0, str(len(arr)))
    for n in arr:
        if not isinstance(n, dict) or not n.get('id'): continue
        miss = REQ_TOP - set(n)
        ok(f"[{n['id'][:6]}·{n.get('status','?')}] 基础字段齐", not miss, str(miss) or 'ok')
        for f in NEW_FIELDS:
            if f not in n:
                ok(f"[{n['id'][:6]}] {f} 字段在", False, '字段缺失')
            else:
                ok(f"[{n['id'][:6]}] {f}={n[f]!r:.40}", True)
        sv = n.get('_sv')
        if sv is None or sv < 1:
            print(f"⚠️  [{n['id'][:6]}] _sv 未迁（={sv!r}）——下次存档时迁移函数会自动补字段")

if __name__ == '__main__':
    main()
```

跑：

```bash
python scripts/dev/_check_note_archive.py path/to/np_notes.json
```

## 8. 注意事项·别踩坑

1. **不动 `NP_AR_STATUS`**：A1 不引新状态，dir 是独立维度，跟状态正交。
2. **不改 `NP_FILM_DOCS_KEY`**：出片台侧的草稿（`xhsFilmDocs`）按篇存 key，不跟档案主键合并。
3. **`_sv` 是「软版本」**：迁移函数幂等，将来 A2~A6 升到 `_sv:2 / 3` 时老档写时自动迁。
4. **`mind.nodes` 上限 30 + sub 子节点上限 12**：超过的截断——别让一份导图把 `np_notes` 一行撑到 100KB+。
5. **口播 mp4 不入 `chrome.storage.local`**：超过 5MB 就爆 STORAGE_QUOTA_BYTES。
6. **不写 `manifest.json`**：IndexedDB 后台 handler 是发 message 协议，不加 manifest key。
