### 1.3 DOM 选择器策略（多 fallback）

小红书主页每篇笔记的卡片底部三联 = 「💬 N  评论 / ⭐ N  fav / ❤️ N  like」（hover 才显示完整文案；但 DOM 上 `.count` 或 `[class*="num"]` 一直在）。

```js
function parseStats(card) {
  const out = { likes: null, favs: null, cmts: null };
  // 方式 1：从 aria-label 拿（最稳）
  for (const el of card.querySelectorAll('[aria-label]')) {
    const al = el.getAttribute('aria-label') || '';
    let m;
    if ((m = al.match(/(\d[\d.,]*[万亿]?)\s*赞/)))  out.likes = toNum(m[1]);
    if ((m = al.match(/(\d[\d.,]*[万亿]?)\s*收藏/))) out.favs  = toNum(m[1]);
    if ((m = al.match(/(\d[\d.,]*[万亿]?)\s*评论/))) out.cmts  = toNum(m[1]);
  }
  // 方式 2：从 .count / [class*="num"] 兜底（按出现顺序：评论 / 收藏 / 点赞 —— 经验顺序）
  if (out.likes === null || out.favs === null || out.cmts === null) {
    const nums = Array.from(card.querySelectorAll('[class*="count"], [class*="num"], [class*="interaction"]'))
      .map(el => (el.textContent || '').trim())
      .map(parseNumLike)        // '1.2万' → 12000；'999' → 999
      .filter(n => n !== null);
    if (out.cmts  === null && nums[0] != null) out.cmts  = nums[0];
    if (out.favs  === null && nums[1] != null) out.favs  = nums[1];
    if (out.likes === null && nums[2] != null) out.likes = nums[2];
  }
  return out;
}

function parseNumLike(s) {
  if (!s) return null;
  const m = s.match(/^([\d.,]+)([万亿]?)$/);
  if (!m) return null;
  const n = parseFloat(m[1].replace(/,/g, ''));
  if (Number.isNaN(n)) return null;
  return Math.round(n * (m[2] === '万' ? 1e4 : m[2] === '亿' ? 1e8 : 1));
}
```

约束：
- 任一字段读不到就留 null，**不要写 0**（写 0 = "0 赞" = 误读）
- 一张卡三联全 null = 卡片结构变了，全条 stat 留 null，落 `note_stats` 时按字段写 null（不要因为 null 整条跳过）
- DOM 选择器走**多 fallback**：先 aria-label（最稳）、再 `.count`/`[class*="num"]`、最后整页 innerText 正则（兜底）

### 1.4 改造点

2. push 笔记时同步 `stats: parseStats(card)`

### 2.2 配对方式（按标题自动 + 手动兜底）

```js
async function npStatsWrite(profile) {
  const today = fmtDay(new Date());
  const stats = (await npStatsRead()) || { byNoteId: {} };
  const notes = await npNotesAll();

  for (const pn of (profile.notes || [])) {
    // 配对：先按精确标题，再按截断 16 字，再按包含
    const hit = notes.find(n =>
      n.title === pn.title ||
      (n.title && pn.title && n.title.slice(0, 16) === pn.title.slice(0, 16)) ||
      (n.title && pn.title && (n.title.includes(pn.title.slice(0, 8)) || pn.title.includes(n.title.slice(0, 8))))
    );
    if (!hit) continue;     // 配不上 → 跳过（不在档案里的笔记不统计）

    const prev = stats.byNoteId[hit.id] || {};
    const day = statusAtToDay(hit.statusAt && hit.statusAt.已发);
    const days = day ? Math.max(1, Math.ceil((Date.now() - day) / 86400e3)) : 1;
    const snap = { likes: pn.stats?.likes ?? null, favs: pn.stats?.favs ?? null, cmts: pn.stats?.cmts ?? null, dayAvg: null };
    snap.dayAvg = (snap.likes != null && days > 0) ? Math.round(snap.likes / days) : null;

    // 同一天重抓 = 覆盖当日末条
    const snapshots = (prev.snapshots || []).filter(s => s.date !== today);
    snapshots.push({ date: today, ...snap });

    stats.byNoteId[hit.id] = {
      likes: snap.likes, favs: snap.favs, cmts: snap.cmts,
      dayAvg: snap.dayAvg, snapAt: Date.now(),
      snapshots,
    };
  }
  await npStatsSave(stats);
}
```

### 2.3 配对不上的人工路径

复盘分析页「已发复盘」里点某条笔记 → 弹「🔗 配对」：

- 选一条匹配 = 改 note_stats 键（或把已发篇 id 写进 note_stats.pairings[抓到的 url] = 档案 id）

## 四、失败兜底文案（沿用现有，不新造）

| 失败原因 | 文案（沿用 extractGuestProfile 已有的） |
|---|---|
