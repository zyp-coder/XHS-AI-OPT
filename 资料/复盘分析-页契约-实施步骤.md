## 3. popup.js·数据层：B12 算法 + npUsageByKbCat

### 3.1 `npUsageByKbCat()` 按知识库类汇总

文件：`extension/popup.js`，**新建**一节，放在 `npUsageIndex` 之后（≈ line `2658`）。

```js
/** 按知识库类汇总「用过的账」：
 *   - 拿到本地知识库所有 active 条目
 *   - 用每条的 cat / id / title 当 key，把档案里出现的「这篇用过这个 id/title」标记成 covered
 *   - 返回 Map<cat, { catTitle, allItems:[{id,title}], coveredIds:Set, blankIds:Set }>
 *   - 没分类时 catTitle = '默认'，所有不分类的进同一个 cat
 *   - 没找到 cat 字段的条目当「未分类」处理（保持兼容：现状知识库条目没 cat 字段） */
async function npUsageByKbCat() {
  const map = new Map();
  const uncat = { catTitle: '默认', allItems: [], coveredIds: new Set(), blankIds: new Set() };
  map.set('默认', uncat);
  try {
    const r = await chrome.runtime.sendMessage({ action: 'getKnowledgeBase' });
    const entries = ((r && r.entries) || []).filter(e => e && e.isActive !== false);
    for (const e of entries) {
      const cat = String(e.cat || '默认').trim() || '默认';
      if (!map.has(cat)) map.set(cat, { catTitle: cat, allItems: [], coveredIds: new Set(), blankIds: new Set() });
      map.get(cat).allItems.push({ id: String(e.id || ''), title: String(e.title || '') });
    }
  } catch (_) {}
  /* 统计 covered：把档案里每条 note.kb 摊进对应的 cat */
  const list = await npNotesAll();
  const hitsPerCat = new Map();
  list.forEach(n => {
    if (!n || !Array.isArray(n.kb)) return;
    for (const k of n.kb) {
      const id = String(k.id || '').trim();
      const ttl = String(k.title || '').trim();
      if (!id && !ttl) return;
      for (const c of map.values()) {
        if (c.allItems.some(it => it.id === id || it.title === ttl)) {
          c.coveredIds.add(id || ttl);
        }
      }
    }
  });
  return map;
}
```

> **驳旧版 `getKnowledgeBase` 没 cat 字段问题**：现状版本拿不到 cat——版本升级前一律归「默认」一类，不要假造。

### 3.2 `npAuditTopicsAlgo()`

```js
/** 选题盘点 = 按知识库类看三类：
 *  - 全讲过：所有条目都被至少一篇档案覆盖 → 三个结论档之一
 *  - 全空白：所有条目都没被任何档案覆盖 → 三个结论档之一
 *  - 有空白 + 有表现：部分覆盖
 *  返回 Array<{ catTitle, allItems, coveredIds, blankIds, conclusion }> */
async function npAuditTopicsAlgo() {
  const map = await npUsageByKbCat();
  const out = [];
  for (const c of map.values()) {
    const blank = c.allItems.filter(it => !c.coveredIds.has(it.id || it.title));
    const covered = c.allItems.filter(it => c.coveredIds.has(it.id || it.title));
    let conclusion;
    if (covered.length === 0) conclusion = '全空白';
    else if (blank.length === 0) conclusion = '全讲过';
    else conclusion = '有空白 + 有表现';
    out.push({
      catTitle: c.catTitle,
      total: c.allItems.length,
      covered: covered.length,
      blank: blank.length,
      conclusion,
      blankIds: blank,
      coveredIds: covered,
    });
  }
  return out;
}
```

### 3.3 `npAuditSentAlgo()`

```js
/** 已发复盘：基于 note_stats（详见上一步骤） */
async function npAuditSentAlgo() {
  const all = await npNotesAll();
  const statsAll = await npStatsReadAll();
  const sent = all.filter(n => n && n.status === '已发');
  const rows = sent.map(n => {
    const eid = (n.url || '').match(/\/explore\/([^?#]+)/);
    const stats = eid ? statsAll[eid[1]] : null;
    return {
      id: n.id,
      title: n.title || '',
      url: n.url || '',
      stats,
      hasStats: !!(stats && (stats.likes || stats.collects || stats.comments)),
      fetchedAt: n.statsAt || (stats && stats.fetchedAt) || 0,
      form: n.form,
      dir: n.dir,
      kOut: n.kOut,
      dOut: n.dOut,
      sentAt: (n.statusAt && n.statusAt['已发']) || (n.updatedAt || 0),
    };
  }).sort((a, b) => b.sentAt - a.sentAt);
  return {
    total: sent.length,
    withStats: rows.filter(r => r.hasStats).length,
    rows,
  };
}
```

```js
async function npAuditTopicsPaint() {
  const pane = document.getElementById('auditTopicsPane');
  if (!pane) return;
  pane.innerHTML = '<div class="audit-loading">📋 选题盘点在算，按知识库类看「讲过 / 没讲过」…</div>';

      tr.addEventListener('click', () => npAuditOpenCat(tr.getAttribute('data-cat')));
    });
  } catch (e) { pane.innerHTML = '❌ 出错了：' + String(e.message || e); }

function npAuditTopicsHtml(cls) {
  if (!cls.length) return '<div class="audit-empty">还没在「诊断&配置 → 知识库」录入条目。先去录入，再回来看盘点。</div>';
  let out = '';
  out += '<table class="audit-tbl">';
  out += '<colgroup><col><col style="width:80px;"><col style="width:80px;"><col style="width:160px;"></colgroup>';
  out += '<thead><tr><th>知识库类</th><th>总条目</th><th>还差</th><th>结论 · 行末查</th></tr></thead><tbody>';
  for (const c of cls) {
    const chip = c.conclusion === '全空白' ? '🟥 全空白'
                : c.conclusion === '全讲过' ? '🟩 全讲过'
                : '🟨 有空白 + 有表现';
    const tip = c.conclusion === '全空白'
        ? '这一类里所有条目都没写过 —— 有料没用上，挑几条加进下次规划'
        : c.conclusion === '全讲过'
        ? '这一类里所有条目都写过 —— 可以换个角度写、或者先停一停'
        : '部分讲过 —— 「还差」列里点进去看具体哪些没写';
    out += `<tr data-cat="${escAttr(c.catTitle)}" title="${escAttr(tip)}">`;
    out += `<td><strong>${esc(c.catTitle)}</strong></td>`;
    out += `<td>${c.total}</td>`;
    out += `<td style="color:${c.blank > 0 ? '#c0392b' : '#8a929e'};">${c.blank}</td>`;
    out += `<td><span class="audit-chip audit-chip-${c.conclusion === '全讲过' ? 'ok' : c.conclusion === '全空白' ? 'bad' : 'mix'}">${chip}</span></td>`;
    out += '</tr>';
    if (c.conclusion === '有空白 + 有表现' && c.blankIds.length) {
      out += `<tr data-cat="${escAttr(c.catTitle)}"><td colspan="4" class="audit-tbl-sub"><b>还没写过：</b>`;
      c.blankIds.slice(0, 8).forEach(it => {
        out += `<span class="audit-blank-item" data-id="${escAttr(it.id)}" data-title="${escAttr(it.title)}">${esc(it.title)}</span> `;
      });
      if (c.blankIds.length > 8) out += `… 等 ${c.blankIds.length} 条`;
      out += '</td></tr>';
    }
  }
  out += '</tbody></table>';
  out += '<div class="audit-hint">点行／点条目 → 这一类的清单列出来；勾上挑一篇发笔记用。</div>';
  return out;
}

async function npAuditOpenCat(cat) {
  // 这一类在档案里出现过的篇 + 还没写过的条目，都一并摆出来
  const list = await npNotesAll();
  const hit = list.filter(n => Array.isArray(n.kb) && n.kb.some(k => String(k.cat || '默认') === cat));
  const map = await npUsageByKbCat();
  const c = map.get(cat) || { blankIds: [] };
  const html = [
    '<div class="audit-cat-detail">',
    '<h3>' + esc(cat) + '</h3>',
    '<h4>已写过：</h4>',
  ];
  if (!hit.length) html.push('<p class="audit-empty">这一类还没写过。</p>');
  else {
    html.push('<ul>');
    hit.forEach(n => {
      html.push('<li><a href="#" data-nid="' + esc(n.id) + '">' + esc(n.title || '（无标题）') + '</a> · ' + (n.status || '') + '</li>');
    });
    html.push('</ul>');
  }
  html.push('<h4>还没写过：</h4>');
  if (!c.blankIds.length) html.push('<p class="audit-empty">全讲过了。</p>');
  else {
    html.push('<ul>');
    c.blankIds.forEach(it => {
      html.push('<li>' + esc(it.title) + (it.id ? ' <small>(' + esc(it.id) + ')</small>' : '') + '</li>');
    });
    html.push('</ul>');
  }
  html.push('</div>');
  /* 用 modal / drawer 简单展示 = 在 popup 里开一个普通 div 浮层 */
  const m = document.createElement('div');
  m.className = 'audit-modal';
  m.innerHTML = html.join('');
  document.body.appendChild(m);
  m.addEventListener('click', (ev) => {
    const t = ev.target.closest('[data-nid]');
    if (t) {
      ev.preventDefault();
      switchMainTab('noteArchive');
      window._npArchiveFocus = t.getAttribute('data-nid');
    } else {
      m.remove();
    }
  });
}
```

```js
function npAuditSentHtml(data) {
  if (!data.total) {
    return [
      '<div class="audit-empty">',
      '还没有「已发」状态的档案可看 —— 先发表几篇，回这里看抓主页的数字趋势。',

    ].join('');

    <span class="chip">已发 ${data.total} 篇 · 抓到 ${data.withStats} 篇（${pct}%）</span>
    <button id="btnStatsRefetch" class="n-btn">🔄 重抓全部已发主页</button>

  </div>`;
  out += '<table class="audit-sent-tbl">';
  out += '<colgroup><col><col style="width:140px;"><col style="width:80px;"><col style="width:80px;"><col style="width:80px;"><col style="width:120px;"></colgroup>';
  out += '<thead><tr><th>标题（点行选中）</th><th>发表日 · 已发天数</th><th>🖊 点赞</th><th>⭐ 收藏</th><th>💬 评论</th><th>抓主页 · 操作</th></tr></thead><tbody>';
  for (const r of data.rows) {
    const age = r.sentAt ? Math.floor((Date.now() - r.sentAt) / 86400e3) + ' 天' : '—';
    const sentDate = r.sentAt ? new Date(r.sentAt).toLocaleDateString() : '—';
    const likes = r.hasStats ? formatNum(r.stats.likes) : '⚪';
    const collects = r.hasStats ? formatNum(r.stats.collects) : '—';
    const comments = r.hasStats ? formatNum(r.stats.comments) : '—';
    const fetchedLabel = r.fetchedAt ? new Date(r.fetchedAt).toLocaleString() : '⚪ 还没抓';
    out += `<tr data-nid="${esc(r.id)}" data-url="${esc(r.url)}" data-title="${escAttr(r.title)}">`;
    out += `<td><button class="btn-open-note">${esc(r.title || '（无标题）')}</button></td>`;
    out += `<td>${sentDate} · ${age}</td>`;
    out += `<td>${likes}</td><td>${collects}</td><td>${comments}</td>`;
    out += `<td><span class="audit-fetched-tip" title="${esc(r.url)}">${fetchedLabel}</span> <button class="btn-stats-one n-btn-sm">🔄</button></td>`;
    out += '</tr>';
  }
  out += '</tbody></table>';
  out += '<div class="audit-hint">💡 单行点中后，「🔄 重抓这一篇」按钮就生效；批量重抓走「🔄 重抓全部」。</div>';
  return out;
}

function formatNum(n) {
  n = Number(n || 0);
  if (!n) return '0';
  if (n < 1000) return String(n);
  if (n < 10000) return (n / 1000).toFixed(1) + 'k';
  return (n / 10000).toFixed(1) + '万';
}
```

### 7.1 `_check_audit_page.py`

文件：`scripts/dev/_check_audit_page.py`，**新建**：

```python
#!/usr/bin/env python3
"""复盘分析页自检：np_notes.json + 知识库导出 json
   - 选中：每个 cat 都有结论
   - 选中：『全空白』的 cat 至少能在界面看到「🟥」chip
   - 选中：『全讲过』的 cat 至少能显示「🟩」chip
"""
import json, pathlib

def main():
    base = pathlib.Path(sys.argv[1] if len(sys.argv) > 1 else '.')
    notes = json.loads((base / 'np_notes.json').read_text(encoding='utf-8'))
    kb_raw = json.loads((base / 'kb.json').read_text(encoding='utf-8'))
    kb = kb_raw if isinstance(kb_raw, list) else (kb_raw.get('entries') or [])
    cats = {}
    for e in kb:
        if not e or e.get('isActive') is False: continue
        c = (e.get('cat') or '默认').strip() or '默认'
        cats.setdefault(c, []).append(e)
    covers = {c: set() for c in cats}
    for n in notes:
        if not isinstance(n, dict) or not n.get('kb'): continue
        for k in n['kb']:
            kid = k.get('id'); ttl = k.get('title')
            for c, items in cats.items():
                if any(it.get('id') == kid or it.get('title') == ttl for it in items):
                    covers[c].add(kid or ttl); break
    print('分类\t总\t覆盖\t空白\t结论')
    for c, items in cats.items():
        cov = covers[c]
        blank = [it for it in items if (it.get('id') or it.get('title')) not in cov]
        conclusion = '全空白' if not cov else ('全讲过' if not blank else '有空白 + 有表现')
        print(f"{c}\t{len(items)}\t{len(cov)}\t{len(blank)}\t{conclusion}")

if __name__ == '__main__':
    main()
```

## 9. 注意事项·别踩坑

1. **不写 demo-note 紫框**（三十改）：复盘页右上原来的紫备注框按口径删除——别又塞回去。
2. **B12 算法不要去扩展 `getKnowledgeBase` 的 schema**：现状版本拿不到 `cat` 字段一律归「默认」，等知识库那边版本升级到带 `cat` 时再走真分类。
3. **不存复盘快照**：`npAuditTopicsAlgo` 每次现算——档案 / 知识库变了它跟着变，别做缓存导致失同步。
4. **stats 7 天过期显示「🕗」**：UI 标的过期图标的「🕗」字样，跟 抓主页契约 §7 一致。
5. **「📋」的复盘表不需要滚动到底刷新**：每行渲染独立的 hover/keydown，滚不到就重新 paint 一遍。
6. **重抓全走的不是开 tab 一次性开多个**：是 `for` 循环单条单条开，免得十几个 tab 同时开被风控盯上。
7. **不带任何 manifest 改动**：buttons/事件/CSS 都不动 manifest.json。
